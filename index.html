function renderIndicesList() { const marketGrid = document.getElementById("marketIndicesGrid"); const commGrid = document.getElementById("commodityIndicesGrid"); const indicesItems = allData.filter(d => d.market === 'INDEX'); const marketCodes = ["KOSPI", "KOSDAQ", "SP500", "NASDAQ", "DJI", "N225", "SSEC"]; const commCodes = ["GOLD", "SILVER", "BRENT", "WTI", "COPPER"]; marketGrid.innerHTML = marketCodes.map(code => { const item = indicesItems.find(d => (d.passed_tags || "").includes(code)) || { name: code, close_price: 0, change_rate: 0 }; const rate = item.change_rate || 0; const isUp = rate >= 0; return `
${item.name} ${item.close_price ? item.close_price.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2}) : '-'} ${isUp ? '+' : ''}${rate.toFixed(2)}%
`; }).join(""); commGrid.innerHTML = commCodes.map(code => { const item = indicesItems.find(d => (d.passed_tags || "").includes(code)) || { name: code, close_price: 0, change_rate: 0 }; const rate = item.change_rate || 0; const isUp = rate >= 0; return `
${item.name} ${item.close_price ? item.close_price.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2}) : '-'} ${isUp ? '+' : ''}${rate.toFixed(2)}%
`; }).join(""); }
