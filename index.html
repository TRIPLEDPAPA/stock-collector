async function fetchLiveMarketData() { if (!isMarketOpen()) { document.getElementById("timerBadge").innerText = "⏳ 휴장일/장마감 (대기중)"; countdown = 30; return; } const now = new Date(); const koreanTime = new Date(now.toLocaleString("en-US", { timeZone: "Asia/Seoul" })); const hours = koreanTime.getHours(); const minutes = koreanTime.getMinutes(); try { let compiledStocks = []; for (let page of [1, 2]) { try { const data = await fetchWithProxy(`https://m.stock.naver.com/api/stocks/marketValue/KOSPI?page=${page}&pageSize=30`); if (data && data.stocks) { data.stocks.forEach(item => { if (isPureStock(item.itemCode, item.stockName)) { compiledStocks.push(parseStockItem(item, "KOSPI")); } }); } } catch(e) {} } for (let page of [1, 2]) { try { const data = await fetchWithProxy(`https://m.stock.naver.com/api/stocks/marketValue/KOSDAQ?page=${page}&pageSize=30`); if (data && data.stocks) { data.stocks.forEach(item => { if (isPureStock(item.itemCode, item.stockName)) { compiledStocks.push(parseStockItem(item, "KOSDAQ")); } }); } } catch(e) {} } if (compiledStocks.length === 0) { throw new Error("네이버 시세 데이터를 불러오지 못했습니다."); } compiledStocks.sort((a, b) => b.trade_amount - a.trade_amount); const indices = fetchMockIndices(); allData = [...compiledStocks.slice(0, 40), ...indices]; const todayStr = koreanTime.toISOString().slice(0, 10); document.getElementById("baseDate").innerText = `기준일자: ${todayStr}`; document.getElementById("updatedTime").innerText = `실시간 갱신됨 (${hours}시 ${minutes}분)`; const totalDouble = allData.filter(d => d.market !== 'INDEX' && d.foreign_net_buy > 0 && d.inst_net_buy > 0).length; document.getElementById("doubleCount").innerText = `(${totalDouble}개)`; renderStockList(); renderIndicesList(); } catch (err) { console.error(err); document.getElementById("cardList").innerHTML = `
실시간 데이터 통신 원활함 확인 중... 잠시 후 재시도됩니다.
`; } finally { countdown = 30; } }
